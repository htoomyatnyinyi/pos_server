### 3. Add `upsertCashRegisters` to `repository.ts`

Add this function near the other upsert functions:

```typescript
// ============================================
// UPSERT CASH REGISTERS
// ============================================
export async function upsertCashRegisters(
  remoteRegisters: any[],
  defaultTenantId: string,
) {
  if (!remoteRegisters.length) return;
  const now = new Date().toISOString();

  const registersToInsert = remoteRegisters.map((r) => ({
    id: r.id,
    remoteId: r.remoteId,
    tenantId: r.tenantId || defaultTenantId,
    storeId: r.storeId,
    name: r.name,
    status: r.status || "CLOSED",
    deletedAt: r.deletedAt,
    syncStatus: "synced",
    syncError: null,
    createdAt: r.createdAt ?? now,
    updatedAt: r.updatedAt ?? now,
    lastSyncedAt: now,
  }));

  await getOfflineDb()
    .insert(cashRegisters)
    .values(registersToInsert)
    .onConflictDoUpdate({
      target: cashRegisters.id,
      set: {
        name: sql`excluded.name`,
        status: sql`excluded.status`,
        storeId: sql`excluded.store_id`,
        deletedAt: sql`excluded.deleted_at`,
        syncStatus: "synced",
        syncError: null,
        updatedAt: sql`excluded.updated_at`,
        lastSyncedAt: now,
      },
    });
}
```

Also add a helper to get local registers:

```typescript
// ============================================
// GET LOCAL CASH REGISTERS
// ============================================
export async function getLocalCashRegisters(storeId?: string) {
  const db = getOfflineDb();
  let query = db.select().from(cashRegisters).$dynamic();

  if (storeId) {
    query = query.where(eq(cashRegisters.storeId, storeId));
  }

  const rows = await query.where(eq(cashRegisters.deletedAt, null));
  return rows.map((row) => toCashRegister(row));
}

export async function getLocalCashRegisterById(id: string) {
  const db = getOfflineDb();
  const [row] = await db
    .select()
    .from(cashRegisters)
    .where(and(eq(cashRegisters.id, id), eq(cashRegisters.deletedAt, null)))
    .limit(1);
  return row ? toCashRegister(row) : undefined;
}
```

### 4. Add `pullCashRegisters` to `syncManager.ts`

In `syncManager.ts`, add:

```typescript
// ============================================
// PULL CASH REGISTERS
// ============================================
async function pullCashRegisters(dispatch: AppDispatch, tenantId: string) {
  try {
    const state = store.getState();
    const token = state.auth?.user?.token;
    if (!token) {
      console.warn("⚠️ No auth token found, skipping cash register pull");
      return { synced: 0 };
    }

    if (!remoteApi.endpoints.getRemoteCashRegisters) {
      console.warn("⚠️ getRemoteCashRegisters endpoint not available");
      return { synced: 0 };
    }

    const { data, error } = await store.dispatch(
      remoteApi.endpoints.getRemoteCashRegisters.initiate(undefined, {
        forceRefetch: true,
      }),
    );

    if (error) {
      if (error.originalStatus === 404 || error.status === "PARSING_ERROR") {
        return { synced: 0 };
      }
      console.error("❌ Cash register pull failed:", error);
      return { synced: 0 };
    }

    const registersData = data?.cashRegisters || data?.data || data || [];
    if (registersData.length > 0) {
      await upsertCashRegisters(registersData, tenantId);
      return { synced: registersData.length };
    }
    return { synced: 0 };
  } catch (error) {
    console.error("❌ Failed to pull cash registers:", error);
    return { synced: 0 };
  }
}
```

Then add it to `syncNow` after pulling stores:

```typescript

const registerResult = await pullCashRegisters(dispatch, tenantId);
```

### 7. Update `processOutboxItem` for `cash_registers`

In `syncManager.ts`, add a case for `"cash_registers"`:

```typescript
case "cash_registers": {
  try {
    if (item.operation === "create") {
      const { data, error } = await store.dispatch(
        remoteApi.endpoints.createRemoteCashRegister.initiate(payload),
      );
      if (error) throw new Error(JSON.stringify(error));
      await db
        .update(cashRegisters)
        .set({ remoteId: data.id, syncStatus: "synced" })
        .where(eq(cashRegisters.id, item.entityId));
    } else if (item.operation === "update") {
      const { error } = await store.dispatch(
        remoteApi.endpoints.updateRemoteCashRegister.initiate({
          id: item.entityId,
          ...payload,
        }),
      );
      if (error) throw new Error(JSON.stringify(error));
      await db
        .update(cashRegisters)
        .set({ syncStatus: "synced" })
        .where(eq(cashRegisters.id, item.entityId));
    } else if (item.operation === "delete") {
      const { error } = await store.dispatch(
        remoteApi.endpoints.deleteRemoteCashRegister.initiate(item.entityId),
      );
      if (error) throw new Error(JSON.stringify(error));
      await db.delete(cashRegisters).where(eq(cashRegisters.id, item.entityId));
    }
    await markOutboxSynced(item.id);
    return { success: true };
  } catch (error) {
    return { success: false, error: (error as Error).message };
  }
}
```
